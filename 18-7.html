Three.js 核心用法速查

API CHEAT-SHEET
1

场景三件套

Scene（容器）+ PerspectiveCamera（观察点）+ WebGLRenderer（画布），缺一不可。

2

几何体与材质

new THREE.BoxGeometry(w,h,d) + MeshPhongMaterial({ color }) → new THREE.Mesh(geo, mat)。

3

交互控制器

引入 OrbitControls.js，new THREE.OrbitControls(camera, renderer.domElement) 即可旋转缩放。

4

光照

至少一盏 AmbientLight 提供环境光 + 一盏 DirectionalLight 模拟太阳方向光。

5

InstancedMesh

大量相同几何体（如散点球）用 InstancedMesh，性能远优于逐个 Mesh。

6

渲染循环

function animate(){ requestAnimationFrame(animate); controls.update(); renderer.render(scene, camera); }

// ① 引入本地 UMD 构建（three.min.js + OrbitControls.js 均挂载到全局 THREE）
//    注意顺序：先 three.min.js，再 OrbitControls.js
<script src="./cdn/three/build/three.min.js"></script>
<script src="./cdn/three/examples/js/controls/OrbitControls.js"></script>

// ② 初始化场景三件套
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0a101c);
const camera = new THREE.PerspectiveCamera(50, aspect, 0.1, 1000);
camera.position.set(8, 7, 10);
const renderer = new THREE.WebGLRenderer({ antialias: true, preserveDrawingBuffer: true });
renderer.setPixelRatio(window.devicePixelRatio);

// ③ OrbitControls 实现鼠标交互
const controls = new THREE.OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.autoRotate = true;

// ④ 柱状图：循环生成 BoxGeometry
for (let i = 0; i < data.length; i++) {
  const h = data[i].value * 0.1;
  const geo = new THREE.BoxGeometry(0.9, h, 0.9);
  const mat = new THREE.MeshPhongMaterial({ color: color });
  const bar = new THREE.Mesh(geo, mat);
  bar.position.set(x, h/2, z);
  scene.add(bar);
}

// ⑤ 渲染循环
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
animate();
